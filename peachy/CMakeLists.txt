# Peachy (PMCP): a standalone C11 library, peachy.dll / libpeachy.so, with one export: peachy_get_api.
# Builds on its own (cmake -S peachy) or from the PCHIE tree (PCHIE_PEACHY=ON). MIT license.
cmake_minimum_required(VERSION 3.16)

  project(peachy C)
  enable_testing()
  set(PEACHY_TOP ON)
else()
  enable_language(C)
  set(PEACHY_TOP OFF)
endif()

  set(PEACHY_SANITIZE "${PCHIE_SANITIZE}")
endif()
set(PEACHY_SANITIZE "${PEACHY_SANITIZE}" CACHE STRING "Peachy sanitizers: address, undefined, thread (comma list), or empty")
option(PEACHY_LIBFUZZER "Build the libFuzzer entry points (clang or clang-cl)" OFF)

set(_pj_san_c "")
set(_pj_san_l "")
if(PEACHY_SANITIZE)
  if(MSVC)
    if(NOT PEACHY_SANITIZE STREQUAL "address")
      message(FATAL_ERROR "PEACHY_SANITIZE: MSVC supports address only")
    endif()
    set(_pj_san_c /fsanitize=address /Zi)
  else()
    set(_pj_san_c -fsanitize=${PEACHY_SANITIZE} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    set(_pj_san_l -fsanitize=${PEACHY_SANITIZE})
  endif()
  message(STATUS "PEACHY_SANITIZE: ${PEACHY_SANITIZE}")
endif()

function(peachy_target tgt strict)
  set_target_properties(${tgt} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(strict)
    set_target_properties(${tgt} PROPERTIES C_EXTENSIONS OFF)
  endif()
  target_include_directories(${tgt} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/src)
  if(MSVC)
    target_compile_options(${tgt} PRIVATE /W4 /utf-8)
    target_compile_definitions(${tgt} PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
  else()
    target_compile_options(${tgt} PRIVATE -Wall -Wextra -Wshadow -Wno-unused-function)
  endif()
  if(_pj_san_c)
    target_compile_options(${tgt} PRIVATE ${_pj_san_c})
    target_compile_definitions(${tgt} PRIVATE PEACHY_CHECKING_BUILD=1)
  endif()
  if(_pj_san_l)
    target_link_options(${tgt} PRIVATE ${_pj_san_l})
  endif()
  if(PEACHY_LIBFUZZER AND NOT MSVC)
    target_compile_options(${tgt} PRIVATE -fsanitize=fuzzer-no-link)
  endif()
endfunction()

function(peachy_link_platform tgt)
  if(WIN32)
    target_link_libraries(${tgt} PRIVATE bcrypt advapi32)
  else()
    target_link_libraries(${tgt} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  endif()
endfunction()

# The library itself: only peachy_get_api is visible.
add_library(peachy SHARED ${PEACHY_SRC})
peachy_target(peachy ON)
peachy_link_platform(peachy)
target_compile_definitions(peachy PRIVATE PEACHY_BUILDING_DLL=1)
set_target_properties(peachy PROPERTIES C_VISIBILITY_PRESET hidden)
if(WIN32)
  set_target_properties(peachy PROPERTIES PREFIX "")
endif()

# The same code as a static library for the in-tree tests, the fuzzers and the host tool's offline commands.
add_library(peachy_core STATIC ${PEACHY_SRC})
peachy_target(peachy_core ON)
set_target_properties(peachy_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(peachy_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(WIN32)
  target_link_libraries(peachy_core PUBLIC bcrypt advapi32)
else()
  target_link_libraries(peachy_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
endif()

# Fuzzing: an in-house mutation fuzzer that runs everywhere, plus libFuzzer entry points when asked.
add_executable(peachy_fuzz fuzz/peachy_fuzz.c)
peachy_target(peachy_fuzz OFF)
target_link_libraries(peachy_fuzz PRIVATE peachy_core)
add_test(NAME peachy_fuzz_smoke COMMAND peachy_fuzz all ${PEACHY_FUZZ_SMOKE_RUNS} 1 ${PEACHY_TEST_TMP})
set_tests_properties(peachy_fuzz_smoke PROPERTIES LABELS "peachy;fuzz" TIMEOUT 900)
if(PEACHY_LIBFUZZER)
  foreach(ft json rpc propose allow)
    add_executable(peachy_libfuzzer_${ft} fuzz/peachy_fuzz.c)
    peachy_target(peachy_libfuzzer_${ft} OFF)
    target_compile_definitions(peachy_libfuzzer_${ft} PRIVATE PEACHY_LIBFUZZER=1 PEACHY_FUZZ_TARGET_${ft}=1)
    target_compile_options(peachy_libfuzzer_${ft} PRIVATE -fsanitize=fuzzer)
    if(MSVC)
      target_link_options(peachy_libfuzzer_${ft} PRIVATE /INCREMENTAL:NO)
    else()
      target_link_options(peachy_libfuzzer_${ft} PRIVATE -fsanitize=fuzzer)
    endif()
    if(MSVC)
      # cl has no fuzzer-no-link, and its coverage hooks need libFuzzer's runtime in the same image, so each MSVC
      # fuzzer compiles the library sources itself (the Visual Studio generator instruments the whole target).
      target_sources(peachy_libfuzzer_${ft} PRIVATE ${PEACHY_SRC})
      target_include_directories(peachy_libfuzzer_${ft} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include
                                 ${CMAKE_CURRENT_SOURCE_DIR}/src)
      target_link_libraries(peachy_libfuzzer_${ft} PRIVATE bcrypt advapi32)
    else()
      target_link_libraries(peachy_libfuzzer_${ft} PRIVATE peachy_core)
    endif()
  endforeach()
endif()

# MSVC's AddressSanitizer runtime is a DLL that sits next to cl.exe and is not on PATH outside a developer prompt.
# Put it on PATH for the tests (and for the processes they start) so a sanitizer build runs from any shell.
if(MSVC AND PEACHY_SANITIZE)
  get_filename_component(_pj_cl_dir "${CMAKE_C_COMPILER}" DIRECTORY)
  get_property(_pj_tests DIRECTORY PROPERTY TESTS)
  if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.22)
    set_tests_properties(${_pj_tests} PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_pj_cl_dir}")
  else()
    string(REPLACE ";" "\\;" _pj_path "${_pj_cl_dir};$ENV{PATH}")
    set_tests_properties(${_pj_tests} PROPERTIES ENVIRONMENT "PATH=${_pj_path}")
  endif()
endif()
